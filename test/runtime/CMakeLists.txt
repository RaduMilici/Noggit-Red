project(LocalRuntimeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Core Network Test Widgets)
add_executable(runtime_tests RuntimeTests.cpp ../../src/noggit/runtime/RuntimeManager.cpp ../../src/noggit/runtime/LocalServerPanel.cpp ../../src/noggit/runtime/LocalClientLauncher.cpp)
