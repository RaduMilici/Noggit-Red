cmake_minimum_required(VERSION 3.16)
project(LocalRuntimeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Core Network Test Widgets)
find_path(MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql mariadb REQUIRED)
find_library(MYSQL_LIBRARY NAMES mysqlclient mariadb REQUIRED)
set(RUNTIME_UI_SOURCES ../../src/noggit/runtime/RuntimeManager.cpp ../../src/noggit/runtime/LocalServerPanel.cpp ../../src/noggit/runtime/LocalClientLauncher.cpp ../../src/noggit/runtime/ClientManager.cpp ../../src/noggit/creator/TestSessionService.cpp ../../src/noggit/creator/Database.cpp ../../src/noggit/creator/ChangeTracker.cpp)
add_executable(runtime_tests RuntimeTests.cpp ${RUNTIME_UI_SOURCES})
target_include_directories(runtime_tests PRIVATE ../../src ${MYSQL_INCLUDE_DIR})
target_compile_definitions(runtime_tests PRIVATE USE_MYSQL_UID_STORAGE)
target_link_libraries(runtime_tests PRIVATE ${MYSQL_LIBRARY} Qt5::Core Qt5::Network Qt5::Test Qt5::Widgets)
enable_testing()
add_test(NAME runtime_tests COMMAND runtime_tests)
set_tests_properties(runtime_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(runtime_smoke RuntimeSmoke.cpp ../../src/noggit/runtime/RuntimeManager.cpp)
target_link_libraries(runtime_smoke PRIVATE Qt5::Core Qt5::Network)
add_executable(parent_death_test ParentDeathTest.cpp)
target_link_libraries(parent_death_test PRIVATE Qt5::Core)
add_test(NAME parent_death_test COMMAND parent_death_test)

add_executable(indicator_test IndicatorTest.cpp ${RUNTIME_UI_SOURCES})
target_include_directories(indicator_test PRIVATE ../../src ${MYSQL_INCLUDE_DIR})
target_compile_definitions(indicator_test PRIVATE USE_MYSQL_UID_STORAGE)
target_link_libraries(indicator_test PRIVATE ${MYSQL_LIBRARY} Qt5::Core Qt5::Network Qt5::Widgets)
add_test(NAME indicator_test COMMAND indicator_test)
set_tests_properties(indicator_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
